Builders Depend on Transformations (Internally)

Builders that need modification steps import and use transformation functions internally:

# matsimpy/builders/nanostructure/twisted.py
from matsimpy.transformation.structural import make_supercell
from matsimpy.transformation.geometric import rotate

def build_twisted_bilayer(bottom: Crystal, top: Crystal, angle: float) -> Crystal:
    supercell = make_supercell(bottom, [n, n, 1])    # uses transformation internally
    rotated_top = rotate(top, angle, [0, 0, 1])       # uses transformation internally
    return stack_layers(supercell, rotated_top)

This is already the pattern today — we're just making it explicit that builders use transformations, not the reverse.

BuilderRegistry

matsimpy/builders/registry.py
@dataclass(frozen=True)
class BuilderSpec:
    name: str                   # "from_prototype_fcc", "build_nanotube"
    category: str               # "bulk" | "surface" | "molecule" | "nanostructure"
    callable: Callable
    description: str
    output_type: type           # Crystal | Molecule
    parameter_schema: dict      # JSON Schema for kwargs
    requires_optional: tuple[str, ...]  # e.g. ("pyxtal",) or ()

class BuilderRegistry:
    _specs: dict[str, BuilderSpec]

    def register(self, spec: BuilderSpec) -> None: ...
    def get(self, name: str) -> BuilderSpec: ...
    def build(self, name: str, **kwargs) -> Structure:
        """Validate kwargs, call constructor, validate output type."""
    def list_by_category(self, category: str) -> list[BuilderSpec]: ...
