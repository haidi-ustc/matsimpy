Storage — Schema / Codec / Backend / Facade

Design Section 4: Separating persistence concerns

Four Components

ComponentFileResponsibility
DocumentEnvelopeschema.pyWrap a core object's dict with metadata (id, created_at, schema_version). Owns reserved-field policy and stable ID generation.
DocumentCodeccodec.pyEncode Structure → DocumentEnvelope, decode back. Separate from backend so serialization doesn't depend on storage engine.
StoreBackendbackend.pyProtocol defining connect/put/get/query/delete/flush/close. MemoryBackend and MaggmaBackend implement it.
DataStoragefacade.pyFacade composing codec + backend. User-facing API: store_data(), retrieve_data(), query(), etc.

1. DocumentEnvelope (schema.py)

matsimpy/storage/schema.py
@dataclass
class DocumentEnvelope:
    """A document ready for persistence. Wraps core payload with metadata."""

    doc_id: str              # stable hash-based ID (content-addressed)
    payload: dict            # the core object's as_dict() output
    schema_version: str      # "1.0.0" — version of the envelope schema
    created_at: str          # ISO 8601 timestamp
    metadata: dict           # user-provided tags/labels/provenance

    # Reserved field names that user metadata must not collide with
    RESERVED_FIELDS = frozenset({"doc_id", "payload", "schema_version",
                                  "created_at", "metadata"})

    @staticmethod
    def generate_id(payload: dict, metadata: dict | None = None) -> str:
        """Content-addressed stable ID: sha256 of canonical JSON."""
        # deterministic: same payload + metadata → same ID

    def to_dict(self) -> dict: ...
    @classmethod
    def from_dict(cls, d: dict) -> DocumentEnvelope: ...

2. DocumentCodec (codec.py)

matsimpy/storage/codec.py
class DocumentCodec:
    """Encodes MatSimPy objects ↔ DocumentEnvelope for persistence."""

    def encode(self, obj: MSONable, metadata: dict | None = None,
               schema_version: str = "1.0.0") -> DocumentEnvelope:
        """Structure → DocumentEnvelope with stable ID."""
        payload = obj.as_dict()
        envelope = DocumentEnvelope(
            doc_id=DocumentEnvelope.generate_id(payload, metadata),
            payload=payload,
            schema_version=schema_version,
            created_at=datetime.utcnow().isoformat(),
            metadata=metadata or {}
        )
        return envelope

    def decode(self, envelope: DocumentEnvelope) -> Structure:
        """DocumentEnvelope → reconstructed Structure (Crystal or Molecule)."""
        return Structure.from_dict(envelope.payload)

3. StoreBackend Protocol (backend.py)

matsimpy/storage/backend.py
from typing import Protocol, runtime_checkable

@runtime_checkable
class StoreBackend(Protocol):
    """Protocol that all storage backends must implement."""

    def connect(self) -> None: ...
    def put(self, doc_id: str, document: dict) -> None: ...
    def get(self, doc_id: str) -> dict | None: ...
    def query(self, criteria: dict, limit: int = 100, offset: int = 0) -> list[dict]: ...
    def delete(self, doc_id: str) -> bool: ...
    def flush(self) -> None: ...
    def close(self) -> None: ...

4. DataStorage Facade (facade.py)

matsimpy/storage/facade.py
class DataStorage:
    """High-level storage API. Composes codec + backend."""

    def __init__(self, backend: StoreBackend | None = None):
        self.codec = DocumentCodec()
        self.backend = backend or MemoryBackend()
        self.backend.connect()

    def store_data(self, obj: MSONable, metadata: dict | None = None) -> str:
        """Encode structure → envelope → persist via backend. Returns doc_id."""
        envelope = self.codec.encode(obj, metadata)
        self.backend.put(envelope.doc_id, envelope.to_dict())
        return envelope.doc_id

    def retrieve_data(self, doc_id: str) -> Structure:
        """Fetch by ID → decode → return Structure."""
        doc = self.backend.get(doc_id)
        if doc is None:
            raise KeyError(f"Document not found: {doc_id}")
        envelope = DocumentEnvelope.from_dict(doc)
        return self.codec.decode(envelope)

    def query(self, criteria: dict, limit=100, offset=0) -> list[Structure]:
        """Query backend → decode results."""
        docs = self.backend.query(criteria, limit, offset)
        return [self.codec.decode(DocumentEnvelope.from_dict(d)) for d in docs]

    def delete(self, doc_id: str) -> bool: ...
    def flush(self) -> None: ...
    def close(self) -> None: ...

Storage Public API (unchanged from user perspective)

# Same usage pattern, new internals
storage = DataStorage()                              # defaults to MemoryBackend
storage = DataStorage(backend=MaggmaBackend(...))    # production
storage = DataStorage(backend=MemoryBackend())       # testing

doc_id = storage.store_data(crystal, metadata={"source": "icsd"})
crystal = storage.retrieve_data(doc_id)
results = storage.query({"metadata.source": "icsd"})
