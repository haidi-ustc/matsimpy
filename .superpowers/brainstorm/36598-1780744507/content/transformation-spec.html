TransformationSpec & Registry

Design Section 2: Structured metadata for every transformation

TransformationSpec Dataclass

matsimpy/transformation/spec.py
@dataclass(frozen=True)
class TransformationSpec:
    """Metadata for a registered transformation."""

    name: str                    # "translate", "create_vacancy", "make_supercell"
    category: str                # "geometric" | "lattice" | "atomic" | "chemical" | "structural"
    callable: Callable           # the actual function
    description: str             # one-line human-readable

    # Type constraints
    applicable_types: tuple      # (Crystal,) | (Crystal, Molecule) | (Molecule,)
    output_type: type            # Crystal | Molecule (usually same as input)

    # Behavior metadata
    preserves_composition: bool  # True for translate/rotate, False for substitution
    preserves_lattice: bool      # True for translate, False for strain/scale
    preserves_site_properties: bool
    preserves_pbc: bool

    # Parameter schema (for validation, UI generation, pipeline introspection)
    parameter_schema: dict       # JSON Schema fragment for kwargs

    # Versioning
    version: str = "1.0.0"       # semver for this spec (not the function)

Examples

Simple: translate()
TransformationSpec(
    name="translate",
    category="geometric",
    callable=translate,
    description="Translate structure by vector",
    applicable_types=(Crystal, Molecule),
    output_type=Crystal,  # or Molecule
    preserves_composition=True,
    preserves_lattice=True,
    preserves_site_properties=True,
    preserves_pbc=True,
    parameter_schema={
        "type": "object",
        "properties": {
            "vector": {
                "type": "array",
                "items": {"type": "number"},
                "minItems": 3, "maxItems": 3
            }
        },
        "required": ["vector"]
    }
)

TransformationRegistry

matsimpy/transformation/registry.py
class TransformationRegistry:
    """Singleton registry for all transformation specs."""

    _specs: dict[str, TransformationSpec]     # name → spec

    def register(self, spec: TransformationSpec) -> None:
        """Register a transformation spec. Raises ValueError on duplicate name."""

    def get(self, name: str) -> TransformationSpec:
        """Look up a spec by name. Raises KeyError if not found."""

    def list_by_category(self, category: str) -> list[TransformationSpec]:
        """Filter specs by category."""

    def list_applicable(self, structure_type: type) -> list[TransformationSpec]:
        """Filter specs applicable to a given structure type."""

    def apply(self, name: str, structure, **kwargs) -> Structure:
        """Look up spec, validate kwargs against parameter_schema, call fn."""

    def apply_plan(self, plan: TransformationPlan, structure) -> Structure:
        """Execute a sequence of registered transformations."""

# Built-in registration happens at module import time
# Plugin entry points: 'matsimpy.transformations' → discover and register()

registry = TransformationRegistry()

Pipeline / Batch / Sweep → TransformationPlan

matsimpy/transformation/composite/plan.py
@dataclass
class TransformationStep:
    spec_name: str          # registry key, e.g. "translate"
    params: dict            # kwargs passed to the callable
    label: str | None       # optional human label

@dataclass
class TransformationPlan:
    """A serializable, reproducible sequence of transformation steps."""
    steps: list[TransformationStep]
    name: str | None
    description: str | None

    def to_dict(self) -> dict: ...
    def from_dict(cls, d: dict) -> TransformationPlan: ...

# Execution strategies (separate from plan definition)
PipelineExecutor.apply(plan, structure)     # sequential apply
BatchExecutor.apply(plan, structures)       # parallel apply
SweepExecutor.apply(plan, structure, sweep_params)  # grid search
